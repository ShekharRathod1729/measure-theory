\section{Types of Convergence}

Let $(\Omega, \F, \mu)$ be a $\sigma$-finite measure space and let $\F$ be $\mu$-complete. \textbf{Recall:} If $f, g: \Omega \to \bar{\R}$ are functions such that $f$ is $\F$-measurable and $g = f$ a.e., then $g$ is also $\F$-measurable.

\textcolor{blue}{We define a relation $\sim$ as follows: $f \sim g$ if $f = g$ a.e.} Clearly, $\sim$ is an equivalence relation. Let $M$ be the set of all $\F$-measurable functions and $\M := M / \sim$, the collection of equivalence classes induced by $\sim$. \\

\textbf{Outline.} We will study the following types of convergence:
\begin{enumerate}
    \item Almost-everywhere convergence
    \item Almost-everywhere uniform convergence
    \item Almost uniform convergence
\end{enumerate}

\subsection{Almost-everywhere convergence}

We first recall the definition of \emph{pointwise convergence}:

\begin{definition}
    Let $f, f_n: E \to \bar{\R}$ such that for all $x \in E$, $f_n(x) \to f(x)$. Then $(f_n)_{n \geq 1}$ \textbf{converges pointwise} to $f$.
\end{definition}

\begin{definition}
    We say \textbf{$f_n$ converges to $f$ almost everywhere} if there exists $E \in \F$ such that $f_n \to f$ pointwise on $E$ and $\mu(E^c) = 0$.
\end{definition}

\begin{lemma}
    Let $f_n \to f$ a.e. and $f_n \sim g_n$ for all $n$. Further, let $f \sim g$. Then, $g_n \to g$ a.e.
\end{lemma}

\begin{proof}
    Since $f_n \to f$ a.e., there exists $A \in \F$ such that $\mu(A^c) = 0$ and $f_n \to f$ pointwise on $A$. Also, since $f \sim g$, there exists $B \in \F$ such that $\mu(B^c) = 0$ and $f(x) = g(x)$ for all $x \in B$.

    Now, $f_n \sim g_n$ and so, for every $n \geq 1$, there is some $E_n \in \F$ with $\mu(E_n^c) = 0$ and $f_n(x) = g_n(x)$ for all $x \in E_n$. 

    \textcolor{red}{Define $$E := A \cap B \cap \bigcap_{n \geq 1} E_n.$$}
    Then, 
    $$E^c = A^c \cup B^c \cup \bigcup_{n \geq 1} E_n^c$$
    and so,
    $$\mu(E^c) \leq \mu(A^c) + \mu(B^c) + \sum_{n \geq 1} \mu(E_n^c) = 0,$$
    i.e., $\mu(E^c) = 0$. Further, since $f_n(x) \to f(x)$ for all $x \in A$ and since $E \sub A$, we have that $f_n(x) \to f(x)$ for all $x \in E$. But for any $x \in E$, $x \in E_n$ for all $n \geq 1$, and so $f_n(x) = g_n(x)$ for all $n \geq 1$. Hence, $g_n(x) \to f(x)$ for all $x \in E$. Lastly we observe that $E \sub B$ and so $f(x) = g(x)$ for all $x \in E$. Thus, $g_n(x) \to g(x)$ for all $x \in E$, and since $\mu(E^c) = 0$, we have $g_n \to g$ a.e.
\end{proof}

\begin{lemma}
    If $f_n \to f$ a.e. and $f_n \to g$ a.e., then $f = g$ a.e.
\end{lemma}

\begin{proof}
    $f_n \to f$ and so for some $E \in \F$, $\mu(E^c) = 0$ and $f_n \to f$ pointwise on $E$. Similarly, $f_n \to g$ pointwise on some $F \in \F$ with $\mu(F^c) = 0$. Define $H := E \cap F$. For all $x \in H$, $f_n(x) \to f(x)$ and $f_n(x) \to g(x)$. By the uniqueness of the limit of a sequence, we have $f(x) = g(x)$ for all $x \in H$. Also, $\mu(H^c) = \mu(E^c \cup F^c) \leq \mu(E^c) + \mu(F^c) = 0$ and so $\mu(H^c) = 0$. Hence, $f = g$ a.e.
\end{proof}

\begin{example}
    Let $\Omega = [0, 1]$, and consider the measure space $(\Omega, \mathscr{L}, \lambda)$, where $\mathscr{L}$ is the Lebesgue $\sigma$-algebra on $\Omega$ and $\lambda$ is the Lebesgue measure. Define $f_n(x) := x^n$. Then, $f_n(x) \to 0$ for all $x \in [0, 1)$ but $f_n(1)$ does not converge to $0$. Hence, $(f_n)_{n \geq 1}$ does not converge pointwise to 0 on $[0, 1]$, but $f_n \to 0$ $\lambda$-a.e.
\end{example}

\subsection{Almost-everywhere uniform convergence}

We recall the definition of \emph{uniform convergence}:

\begin{definition}
    Let $f, f_n: E \to \bar{\R}$. We say that \textbf{$f_n$ converges to $f$ uniformly} if $\forall \epsilon > 0$, there exists $n_0 \in \mathbb{N}$ such that for all $n \geq n_0$, $|f_n(x) - f(x)| < \epsilon$ for all $x \in E$; or, equivalently, for all $n \geq n_0$, 
    $$ \sup_{x \in E} |f_n(x) - f(x)| < \epsilon.$$
\end{definition}

\begin{definition}
    Let $f, f_n: \Omega \to \bar{\R}$. Then, \textbf{$f_n$ converges to $f$ uniformly almost everywhere} if there exists some $E \in \F$ such that $\mu(E^c) = 0$ and $f_n \to f$ uniformly on $E$.
\end{definition}

\begin{definition}
    Let $f: \Omega \to \bar{\R}$ be a function and suppose there exists some $a > 0$ such that $\mu \{ x: |f(x)| > a \} = 0$. We say that $f$ is \textbf{essentially bounded} and define the \textbf{essential supremum} of $f$ by
    $$ \textup{ess sup } |f| = \inf \{ a > 0 : \mu \{ x:|f(x)| > a \} = 0\}. $$ 
\end{definition}

Clearly, $\essup |f| \leq \sup |f|$.

\begin{example}
    Consider the following function:
    $$ f(x) = 
    \begin{cases}
    1, & \text{if } x \neq 0, \\
    2, & \text{if } x = 0.    
    \end{cases} $$
    Then, $\essup |f| = 1$ while $\sup |f| = 2$.
\end{example}

\begin{proposition}
    If $\essup |f| = c$, then $\mu \{ x : |f(x)| > c \} = 0$.
\end{proposition}

\begin{proof}
    Since $c = \essup |f|$, we have $c = \inf \{ a > 0 : \mu \{ x:|f(x)| > a \} = 0\}$, and so for any $d > c$, $\mu \{ x:|f(x)| > d \} = 0$. Hence, for all $n \in \N$, $\mu \left\{ x:|f(x)| > c + \frac{1}{n} \right\} = 0$. Therefore, 
    $$\mu \left( \bigcup_{n \geq 1}\left\{ x:|f(x)| > c + \frac{1}{n} \right\}\right) = 0,$$ 
    that is, $\mu \{ x : |f(x)| > c \} = 0$.
\end{proof}

\begin{proposition}
    The essential supremum is a class property: if $f \sim g$, then $ \essup |f| = \essup |g|$.
\end{proposition}

\begin{proof}
    Let $c = \essup |f|$ and $A = \{ x : |g(x)| > c \}$. Since $f \sim g$, there exists some $E \in \F$ such that $\mu(E^c) = 0$ and $f(x) = g(x)$ for all $x \in E$.

    \textcolor{red}{Now, $ A = (A \cap E) \cup (A \cap E^c) $ and so $\mu(A) = \mu(A \cap E) + \mu(A \cap E^c)$.} But $A \cap E^c \sub E^c$ and since $\mu(E^c) = 0$, $\mu(A \cap E^c) = 0$ and so $\mu(A) = \mu(A \cap E)$.

    Also, $A \cap E = \{ x \in E : |g(x)| > c \} = \{ x : |f(x)| > c \}$ since $f = g$ on $E$. Hence, $\mu(A \cap E) = \mu \{ x : |f(x)| > c \} = 0$ by Proposition 17.10, and so $\mu(A) = 0$, i.e., $\mu\{ x : |g(x)| > c \} = 0$. Therefore, $\essup |g| \leq c = \essup |f|$. 
    
    It can be similarly shown that $\essup |f| \leq \essup |g|$ and so $\essup |f| = \essup |g|$.
\end{proof}

Let $\LI$ be the subset of $\mathscr{M}$ containing all the essentially bounded functions in $\mathscr{M}$. 

\begin{lemma}
    Let $[f], [g]$ be equivalence classes in $\LI$. Define $d([f], [g]) := \essup |f - g|$. Then, $d$ is a metric on $\LI$.
\end{lemma}

\begin{proof}
    Clearly, $d([f], [g]) \geq 0$ and $d([f], [g]) = d([g], [f])$. 

    \textbf{We first show:} if $d([f], [g]) = 0$, then $[f] = [g]$, i.e., $f \sim g$. Define $h := f - g$. Since $d([f], [g]) = 0$, $ \essup |h| = 0$. Hence, for all $n \in \N$, 

    $$ \mu \left\{ x : |h(x)| > \frac{1}{n} \right\} = 0 ,$$
    and so
    $$ \mu \left( \bigcup_{n \geq 1} \left\{ x : |h(x)| > \frac{1}{n} \right\} \right) = 0,$$
    that is, 
    $$ \mu \left\{ x : |h(x)| > 0 \right\} = 0. $$
    Therefore, $|h(x)| = 0$ a.e., and so $|f(x) - g(x)| = 0$ a.e., implying that $f(x) = g(x)$ a.e., or that $f \sim g$. 

    \textbf{We now show that $d([f], [h]) \leq d([f], [g]) + d([g], [h])$.} Let $d_1 := d([f], [g])$ and $d_2 := d([g], [h])$. We want to show that $\essup |f - h| \leq d_1 + d_2$. 
    
    \textcolor{red}{Consider the set $A = \{ x: |f(x) - h(x)| > d_1 + d_2 \}$.} For all $x \in A$, 

    $$ d_1 + d_2 < |f(x) - h(x)| \leq |f(x) - g(x)| + |g(x) - h(x)| $$
    which is only possible if either $d_1 < |f(x) - g(x)|$ or $d_2 < |g(x) - h(x)|$ or both (for if $|f(x) - g(x)| \leq d_1$ and $|g(x) - h(x)| \leq d_2$ then $|f(x) - g(x)| + |g(x) - h(x)| \leq d_1 + d_2$). Hence, $A \sub \{ x : |f(x) - g(x)| > d_1 \} \cup \{ x : |g(x) - h(x)| > d_2 \}$. Then, 
    $$\mu(A) \leq \mu\{ x : |f(x) - g(x)| > d_1 \} + \mu\{ x : |g(x) - h(x)| > d_2 \} = 0$$
    for $d_1 = \essup |f - g|$ and $d_2 = \essup |g - h|$. Therefore, $\mu\{ x: |f(x) - h(x)| > d_1 + d_2 \} = 0$, and so $\essup |f - h| \leq d_1 + d_2$, as desired.
\end{proof}

\begin{theorem}
    $f_n \to f$ uniformly almost everywhere if and only if $d([f_n], [f]) \to 0$. 
\end{theorem}

\begin{proof}
    Suppose $f_n \to f$ uniformly a.e. Then, $f_n \to f$ uniformly on some $E \in \F$ with $\mu(E^c) = 0$. Let $\epsilon > 0$ and $\delta = \epsilon / 2$. There exists some $N \in \N$ such that, for all $n \geq N$, 
    $$ \sup_{x \in E} |f_n(x) - f(x)| < \delta. $$
    \textcolor{red}{Observe that, for $n \geq N$, $ \{ x : |f_n(x) - f(x)| \geq \delta \} \sub E^c$.} Hence, $\mu \{ x : |f_n(x) - f(x)| \geq \delta \} = 0$, and so, $\essup |f_n - f| \leq \delta$, i.e., $d([f_n], [f]) \leq \delta < \epsilon$, for all $n \geq N$. Thus, $d([f_n], [f]) \to 0$.

    We now show the reverse direction: let $d([f_n], [f]) \to 0$, i.e., $\essup |f_n - f| \to 0$. Then, for every $\epsilon > 0$, there exists some $N \in \N$ such that for all $n \geq N$, $\essup |f_n - f| < \epsilon$, i.e., $\mu \{ x : |f_n - f| \geq \epsilon \} = 0$. Therefore, for every $k \in \N$, there exists some $N_k \in \N$ with 
    $$ \mu \left \{ x : |f_n(x) - f(x)| \geq \frac{1}{2^k} \right \} = 0 $$
    for all $n \geq N_k$, and so
    $$ \mu \left( \bigcup_{n \geq N_k} \left \{ x : |f_n(x) - f(x)| \geq \frac{1}{2^k} \right \} \right) = 0 .$$
    Since this is true for every $k \geq 1$, 
    $$ \mu \left( \bigcup_{k \geq 1} \bigcup_{n \geq N_k} \left \{ x : |f_n(x) - f(x)| \geq \frac{1}{2^k} \right \} \right) = 0 .$$
    \textcolor{red}{Define $E^c := \bigcup_{k \geq 1} \bigcup_{n \geq N_k} \left \{ x : |f_n(x) - f(x)| \geq \frac{1}{2^k} \right \}$.} Then, $E = \bigcap_{k \geq 1} \bigcap_{n \geq N_k} \left \{ x : |f_n(x) - f(x)| < \frac{1}{2^k} \right \}$. We now show that $f_n \to f$ uniformly a.e. on $E$. 

    Let $\epsilon > 0$. Consider $k_0$ such that $\frac{1}{2^{k_0}} < \epsilon$. For all $n \geq N_{k_0}$ and for all $x \in E$, we have $|f_n(x) - f(x)| < \frac{1}{2^{k_0}} < \epsilon$. Equivalently, for all $n \geq N_{k_0}$, $\sup_{x \in E} |f_n(x) - f(x)| < \epsilon$, and since $\mu(E^c) = 0$, $f_n \to f$ uniformly a.e.

\end{proof}

\begin{example}
    Consider the measure space $([0,1], \L, \lambda)$, where $\L$ is the Lebesgue $\sigma$-algebra on $[0,1]$ and $\lambda$ is the Lebesgue measure. Let $f_n(x) = x^n$ and $f = 0$. Then, $f_n \to f$ a.e. but $f_n$ does not converge to $f$ uniformly almost everywhere. To see why this is the case, suppose $f_n \to f$ uniformly a.e. Then, there exists $E \in \L$ such that $\lambda(E^c) = 0$ and $f_n \to f$ uniformly on $E$. Let $0 < \epsilon < 1$. There exists some $N \in \N$ such that, for all $n \geq N$, 
    $$ \sup_{x \in E} |f_n(x) - f(x)| < \epsilon, $$
    i.e., 
    $$ \sup_{x \in E} x^n < \epsilon.$$
    Since this is true for all $n \geq N$, this is true for $N$, and so $\sup_{x \in E} x^N < \epsilon$. But observe that $x^N \geq \epsilon$ for all $x \geq \epsilon^{1/N}$. Hence, if $x \geq \epsilon^{1/N}$, then $x \in E^c$. That is, $\{ x : x \geq \epsilon^{1/N} \} \sub E^c$. But $\lambda \{ x : x \geq \epsilon^{1/N} \} = 1 - \epsilon^{1/N} > 0$ while $\lambda(E^c) = 0$, which is a contradiction. 
\end{example}

\subsection{Almost uniform convergence}

\begin{definition}
    Let $f_n, f : \Omega \to \bar{\R}$. We say that $f_n$ converges to $f$ \textbf{almost uniformly} if for every $\epsilon > 0$, there exists $E_\epsilon \in \F$ such that $\mu(E_\epsilon^c) < \epsilon$ and $f_n \to f$ uniformly on $E_\epsilon$.
\end{definition}

\begin{example}
    Consider once again the function $f_n(x) = x^n$ and $f = 0$ on $[0, 1]$. For any $\epsilon \in (0, 1)$, define $E_\epsilon := [0, 1 - \epsilon)$. Then, $E_\epsilon^c = [1 - \epsilon, 1]$ and so $\lambda(E^c) = \epsilon$. Also, $\sup_{x \in E_\epsilon} |x^n - 0| < (1 - \epsilon)^n \to 0$ and so $f_n \to f$ uniformly on $E_\epsilon$. Thus, $f_n \to f$ almost uniformly.
\end{example}

\begin{remark}
    Almost-everywhere uniform convergence is the strongest: almost-everywhere uniform convergence implies almost-everywhere convergence and almost uniform convergence. 
\end{remark}

We now show that almost uniform convergence implies almost-everywhere convergence.

\begin{lemma}
    If $f_n \to f$ almost uniformly, then $f_n \to f$ a.e.
\end{lemma}

\begin{proof}
    Since $f_n \to f$ almost uniformly, for all $k \geq 1$, there exists some $E_k \in \F$ such that $f_n \to f$ uniformly on $E_k$ and $\mu(E_k^c) < 1/k$. \textcolor{red}{Define $E = \bigcup_{k \geq 1} E_k$.} Then, $E^c = \bigcap_{k \geq 1} E_k^c \sub E_{k_0}^c$ for all $k_0 \in \N$ and so $\mu(E^c) \leq \mu(E_{k_0}^c) < 1/k_0$ for all $k_0 \in \N$, i.e., $\mu(E^c) = 0$. Also, for any $x \in E$, we have $x \in E_k$ for some $k \geq 1$, and since $f_n \to f$ uniformly on $E_k$, $f_n(x) \to f(x)$. Hence, for all $x \in E$, $f_n \to f$ and $\mu(E^c) = 0$, i.e., $f_n \to f$ a.e.
\end{proof}

\begin{theorem}
    (\textbf{Egorov's theorem}.) Let $\mu(\Omega) < \infty$. Then, if $f_n \to f$ a.e., then $f_n \to f$ almost uniformly. 
\end{theorem}

\begin{proof}
    \textcolor{red}{Observe that since $f_n \to f$ a.e., the set $E$ on which $f_n \to f$ is given by:
    $$ E = \bigcap_{k \geq 1}\bigcup_{n \geq 1}\bigcap_{l \geq n} \left\{x : |f_l(x) - f(x)| < \frac{1}{k} \right\},$$
    where $\mu(E^c) = 0$.} Hence,
    $$ \mu \left( \bigcup_{k \geq 1}\bigcap_{n \geq 1}\bigcup_{l \geq n} \left\{x : |f_l(x) - f(x)| \geq \frac{1}{k} \right\} \right) = 0 .$$
    Then, for all $k \geq 1$, 
    $$ \mu \left( \bigcap_{n \geq 1}\bigcup_{l \geq n} \left\{x : |f_l(x) - f(x)| \geq \frac{1}{k} \right\} \right) = 0 .$$
    Fix $k$ and define $A_n = \bigcup_{l \geq n} \left\{x : |f_l(x) - f(x)| \geq \frac{1}{k} \right\}$. Clearly, $A_{n + 1} \sub A_n$ and so let $A_n \downarrow \bigcap_{m \geq 1} A_m$. Since $\mu(\Omega) < \infty$, we use continuity from above:
    $$ \mu(A_n) \downarrow \mu\left( \bigcap_{m \geq 1} A_m \right) = 0 $$
    and so, for every $k \geq 1$,  
    $$ \lim_{n \to \infty} \mu \left( \bigcup_{l \geq n} \left\{x : |f_l(x) - f(x)| \geq \frac{1}{k} \right\} \right) = 0.$$
    Let $\epsilon > 0$. For a fixed $k$, there exists $n_{\epsilon, k}$ such that for all $n \geq n_{\epsilon, k}$, 
    $$ \mu \left( \bigcup_{l \geq n_{\epsilon, k}} \left\{x : |f_l(x) - f(x)| \geq \frac{1}{k} \right\} \right) < \frac{\epsilon}{2^k}.$$
    Hence, 
    \begin{align*}
    \mu \left( \bigcup_{k \geq 1}\bigcup_{l \geq n_{\epsilon, k}} \left\{x : |f_l(x) - f(x)| \geq \frac{1}{k} \right\} \right) 
    &\leq \sum_{k \geq 1} \mu \left( \bigcup_{l \geq n_{\epsilon, k}} \left\{x : |f_l(x) - f(x)| \geq \frac{1}{k} \right\} \right) \\
    &< \sum_{k \geq 1} \frac{\epsilon}{2^k} = \epsilon. 
    \end{align*}
    Define $E_\epsilon := \bigcap_{k \geq 1}\bigcap_{l \geq n_{\epsilon, k}} \left\{x : |f_l(x) - f(x)| < \frac{1}{k} \right\}$. We just showed that $\mu(E_\epsilon^c) < \epsilon$. We now show that $f_n \to f$ uniformly on $E_\epsilon$. 
    
    For any $\delta > 0$, take $k_0 \in \N$ with $\frac{1}{k_0} < \delta$. For all $l \geq n_{\epsilon, k_0}$, we have $|f_l(x) - f(x)| < 1/k_0 < \delta$ for all $x \in E_\epsilon$. Hence, $f_n \to f$ uniformly on $E_\epsilon$. Thus we have shown that $f_n \to f$ almost uniformly. 
\end{proof}
