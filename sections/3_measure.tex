\section{Measure}

Let $\Omega$ be a set. 

\begin{definition}
  Let $\F \sub \Po(\Omega)$ be a $\sigma$-algebra, and let $\mu : \F \to [0, +\infty]$ be a function such that:
  
  (1) $\mu(\varnothing) = 0$
  
  (2) (\textbf{Countable additivity} or \textbf{$\sigma$-additivity}.) If $(E_j)_{j \geq 1}$ is a sequence of of pairwise disjoint sets in $\F$, then
  \[
  \mu \left( \bigsqcup_{j \geq 1} E_j \right) = \sum_{j \geq 1} \mu(E_j).
  \]

  Then, $\mu$ is called a \textbf{measure}.
\end{definition}

Countable additivity implies \textbf{finite additivity}: If $E_1,...,E_n$ is a sequence of pairwise disjoint sets in $\F$, then
\[
\mu \left( \bigsqcup_{j = 1}^n E_j \right) = \sum_{j = 1}^n \mu(E_j).
\]

\begin{proposition}
  Let $E, F \in \F$ such that $E \sub F$ and let $\mu$ be a measure on $\F$. Then, $\mu(E) \leq \mu(F)$. Further, if $\mu(E) < +\infty$, then $\mu(F \sm E) = \mu(F) - \mu(E)$. 
\end{proposition}
\begin{proof}
  Since $E \sub F$, we have $F = (E) \sqcup (F \sm E)$, and so $\mu(F) = \mu(E) + \mu(F \sm E)$. Since $\mu(F \sm E)$ is nonnegative, $\mu(F) \geq \mu(E)$. 

  Now, let $\mu(E) < +\infty$. Since $\mu(F) = \mu(E) + \mu(F \sm E)$ and $\mu(E) < + \infty$, we can subtract $\mu(E)$ from both sides to obtain $\mu(F \sm E) = \mu(F) - \mu(E)$. 
\end{proof}

\begin{example}
  (\textbf{Discrete measure.}) Let $\F \sub \Po(\Omega)$ be a $\sigma$-algebra. Let $x_1, x_2,... \in \Omega$ and $p_1, p_2, ... \geq 0$. For all $A \in \F$, define
  \[
  \mu(A) := \sum_{j \geq 1} p_j \mathbf{1}_A(x_j)
  \]
  where $\mathbf{1}$ is the indicator function: 
  \[
  \mathbf{1}_A(x_j) =
  \begin{cases}
    1, \text{ if } x_j \in A \\
    0, \text{ if } x_j \not\in A 
  \end{cases}.
  \]
  Then, $\mu$ is a measure. 
\end{example}

\begin{example}
  Let $\Omega = (0, 1)$ and $\C := \{ (a, b] : 0 \leq a < b < 1 \}$. Define $\mu: \C \to \nonneg$ by
  \[ 
  \mu((a, b]) = 
  \begin{cases}
  + \infty, \text{ if } a = 0 \\
  b - a, \text{ if } a > 0
  \end{cases}.
  \]

  Then, $\mu$ is finitely additive, but not countably additive. To see this, consider a strictly monotonic decreasing sequence $x_n \to 0$ where $x_1 = \frac{1}{2}$. Then, 
  \[
  \bigsqcup_{j \geq 1} (x_{j + 1}, x_j] = \left(0, \frac{1}{2}\right]
  \]
  and so $\mu\left(\bigsqcup_{j \geq 1} (x_{j + 1}, x_j]\right) = 0$, but $\sum_{j \geq 1} \mu((x_{j + 1}, x_j]) = \sum_{j \geq 1} (x_j - x_{j + 1}) = (x_1 - x_2) + (x_2 - x_3) + ... = \frac{1}{2}$.
\end{example}

\begin{lemma}
  Let $\mu$ be a measure on a $\sigma$-algebra $\F$ and $E_n, E \in \F$ for all $n \geq 1$.

  (1) (\textbf{Continuity from below.}) If $E_n \uparrow E$, then $\lim_{n \to \infty} \mu(E_n) = \mu(E)$.
  
  (2) (\textbf{Continuity from above.}) If $E_n \downarrow E$ and $\mu(E_{N}) < + \infty$ for some $N \in \mathbb{N}$, then $\lim_{n \to \infty} \mu(E_n) = \mu(E)$.
\end{lemma}

\begin{proof}
  (1) We have $E_1 \sub E_2 \sub ... \sub E$ and $\bigcup_{j \geq 1} E_j = E$. \textcolor{red}{Define $F_1 := E_1$ and $F_j := E_j \sm E_{j - 1}$ for $j \geq 2$.} Then, $(F_j)_{j \geq 1}$ is a disjoint sequence of sets. Further, 
  \[
  \bigsqcup_{j \geq 1} F_j = \bigcup_{j \geq 1} E_j = E
  \]
  and so
  \[
  \mu \left( \bigsqcup_{j \geq 1} F_j\right) = \mu(E).
  \]
  Since $\mu$ is a measure, 
  \[
  \sum_{j \geq 1} \mu(F_j) = \mu(E). \tag{*}
  \]
  Observe that $\bigsqcup_{j = 1}^n F_j = F_1 \sqcup F_2 \sqcup ... \sqcup F_n = (E_1) \sqcup (E_2 \sm E_1) \sqcup ... \sqcup (E_n \sm E_{n - 1})= E_n$ and so
  \[
  \mu \left(\bigsqcup_{j = 1}^n F_j \right) = \mu(E_n),
  \]
  that is, 
  \[
  \sum_{j = 1}^n \mu(F_j) = \mu(E_n).
  \]
  Taking the limit as $n \to \infty$ on both sides, 
  \[
  \sum_{j \geq 1} \mu(F_j) = \lim_{n \to \infty} \mu(E_n).
  \]
  Combining this with (*) gives $\lim_{n \to \infty} \mu(E_n) = \mu(E)$, as desired. 

  (2) We have $E_1 \supseteq E_2 \supseteq ... \supseteq E$ and $\bigcap_{j \geq 1} E_j = E$. \textcolor{red}{Define $G_1 := E_{N} \sm E_{N + 1}$, $G_2 := E_{N} \sm E_{N + 2}$, and so on.} 
  
  Since $E_n \downarrow E$, we have $E_{N + n} \downarrow E$, and so $(E_N \sm E_{N + n}) \uparrow (E_N \sm E)$, i.e., $G_n \uparrow (E_N \sm E)$. By (1), $\lim_{n \to \infty} \mu(G_n) = \mu(E_N \sm E)$. Since $E \sub E_N$, by Proposition 3.2 we have $\mu(E) \leq \mu(E_N)$. But $\mu(E_N) < +\infty$ and so $\mu(E) < + \infty$. By Proposition 3.2, we get that $\mu(E_N \sm E) = \mu(E_N) - \mu(E)$. Similarly, $\mu(G_n) = \mu(E_N \sm E_{N + n}) = \mu(E_N) - \mu(E_{N + n})$. Hence, 
  \[ \lim_{n \to \infty} \mu(E_N) - \mu(E_{N + n}) = \mu(E_N) - \mu(E) \]
  and so $\lim_{n \to \infty} \mu(E_{N + n}) = \mu(E)$, i.e., $\lim_{n \to \infty} \mu(E_n) = \mu(E)$. 
\end{proof}
